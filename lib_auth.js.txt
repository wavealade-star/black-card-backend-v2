
const jwt = require('jsonwebtoken');
const supabase = require('./supabase');

const JWT_SECRET = process.env.JWT_SECRET || 'black-card-secret';

function generateToken(user, role = 'user') {
  return jwt.sign({ id: user.id, email: user.email, role }, JWT_SECRET, { expiresIn: '7d' });
}

function generateReferralCode() {
  return 'BC' + Math.random().toString(36).substring(2, 8).toUpperCase();
}

async function authMiddleware(req, res, next) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  if (!token) return res.status(401).json({ error: 'No token, access denied' });
  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    // Check if user still exists and not suspended
    if (decoded.role === 'user') {
      const { data: user } = await supabase.from('users').select('id, is_suspended, is_active, deleted_at').eq('id', decoded.id).single();
      if (!user || user.is_suspended || !user.is_active || user.deleted_at) {
        return res.status(403).json({ error: 'Account suspended or deleted' });
      }
    }
    req.user = decoded;
    next();
  } catch (e) {
    return res.status(401).json({ error: 'Invalid token' });
  }
}

function ownerOnly(req, res, next) {
  if (req.user.role !== 'owner') return res.status(403).json({ error: 'Owner only' });
  next();
}

function adminOnly(req, res, next) {
  if (!['owner','support'].includes(req.user.role)) return res.status(403).json({ error: 'Admin only' });
  next();
}

module.exports = { generateToken, generateReferralCode, authMiddleware, ownerOnly, adminOnly, JWT_SECRET };
